def preprocess(x, window=11):
    # remove baseline drift
    baseline = snv(x)
    smooth = savgol_filter(
        baseline,
        window_length=window,
        polyorder=2
    )
    return smooth

spectrum = load_sample("sample_014.csv")
clean = preprocess(spectrum)
